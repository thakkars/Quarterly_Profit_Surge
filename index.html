/* =================================================== BUILD CHARTS GRID Uses TradingView symbol-overview widget with chartOnly:true — works for NSE India stocks on third-party websites. Clicking the chart opens the full TradingView chart in a new tab. =================================================== */ function renderChartsGrid() { const grid = document.getElementById('charts-grid'); const countEl = document.getElementById('charts-count'); if (!currentFilteredData || currentFilteredData.length === 0) { grid.innerHTML = '
No stocks match the current filters.
'; countEl.textContent = '0 stocks'; return; } const nseStocks = currentFilteredData.filter(r => (r["NSE Code Exact"] || "").toString().trim() !== ""); const bseOnly = currentFilteredData.filter(r => (r["NSE Code Exact"] || "").toString().trim() === ""); countEl.textContent = `${currentFilteredData.length} stocks (${nseStocks.length} with charts)`; grid.innerHTML = ''; nseStocks.forEach((row, idx) => { const nse = (row["NSE Code Exact"] || "").toString().trim(); const symbol = `NSE:${nse}`; const name = (row["Company (Page)"] || row["Name"] || nse).toString(); const screener = (row["Screener Name Exact"] || row["Name"] || "").toString().trim(); const industry = (row["Industry"] || "").toString(); const surge = (row["70% Surge"] || "").toString(); const containerId = `tv_ov_${nse}_${idx}`; const scrLink = screener ? `SCR` : ''; const tvNewTab = `TV ↗`; const card = document.createElement('div'); card.className = 'chart-card'; card.id = `card_${nse}`; card.innerHTML = `
${name}
${symbol}  |  ${industry}${surge ? '  |  Surge: ' + surge + '' : ''}
${scrLink}${tvNewTab}
`; grid.appendChild(card); const script = document.createElement('script'); script.type = 'text/javascript'; script.src = 'https://s3.tradingview.com/external-embedding/embed-widget-symbol-overview.js'; script.async = true; script.innerHTML = JSON.stringify({ "symbols": [[name, `${symbol}|12M`]], "chartOnly": false, "width": "100%", "height": 300, "locale": "in", "colorTheme": "light", "autosize": true, "showVolume": true, "showMA": true, "hideDateRanges": false, "hideMarketStatus": false, "hideSymbolLogo": false, "scalePosition": "right", "scaleMode": "Normal", "fontFamily": "-apple-system, BlinkMacSystemFont, Trebuchet MS, Roboto, Ubuntu, sans-serif", "fontSize": "10", "noTimeScale": false, "valuesTracking": "1", "changeMode": "price-and-percent", "chartType": "candlesticks", "lineType": 0, "upColor": "#22ab94", "downColor": "#f7525f", "borderUpColor": "#22ab94", "borderDownColor": "#f7525f", "wickUpColor": "#22ab94", "wickDownColor": "#f7525f", "largeChartUrl": `https://in.tradingview.com/chart/?symbol=${encodeURIComponent(symbol)}` }); document.getElementById(`container_${containerId}`).appendChild(script); }); if (bseOnly.length > 0) { const warn = document.createElement('div'); warn.style.cssText = 'grid-column:1/-1;background:#fff5f5;border:1px solid #fed7d7;border-radius:8px;padding:12px 16px;color:#c53030;font-size:13px;font-weight:600;margin-top:8px;'; warn.innerHTML = `⚠️ ${bseOnly.length} BSE-only stock(s) — no chart: ${bseOnly.map(r => (r["Company (Page)"] || r["Name"] || "")).join(', ')}`; grid.appendChild(warn); } }
